Published formula (Module D & F)

rating = Low | Medium | High per Soil Health Card band

dose_factor = 1.25 (Low) | 1.00 (Medium) | 0.75 (High)

dose_per_acre = crop_RDF_per_ha ÷ 2.471 × dose_factor

P_carrier = DAP, or SSP when DAP's own N would overshoot

carrier_kg = P2O5_required ÷ carrier_P_content

urea_kg = (N_required − carrier_kg × carrier_N) ÷ 0.46

mop_kg = K2O_required ÷ 0.60

gross_depth_mm = soil_depth_mm ÷ application_efficiency

litres_per_turn = gross_depth_mm × 4046.86 × acres

water_saved = flood_season_litres − season_litres

co2_saved = fertiliser_saved × product_CO2 + water_saved × 0.0004

score = irrigation(35) + carbon(25) + pH(20) + precision(20)
